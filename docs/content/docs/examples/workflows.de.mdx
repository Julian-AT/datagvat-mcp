# Vollständige Workflows

Lernen Sie, wie Sie Austria MCP Tools für vollständige Datenentdeckungs-Workflows kombinieren.

## Workflow 1: Suchen → Vorschau → Download

Vollständige Reise von der Suche bis zum Datenzugriff.

### Schritt 1: Nach Datensätzen suchen

```python
# Gesundheitsbezogene CSV-Datensätze aus Wien finden
ergebnisse = search_datasets(
  query="Gesundheitsstatistiken",
  themes=["HEAL"],
  formats=["CSV"],
  publishers=["Stadt Wien"],
  quality_boost=True,
  limit=10
)

# Anzahl der Ergebnisse anzeigen
print(f"{ergebnisse['count']} Datensätze gefunden")

# Top 3 Ergebnisse überprüfen
for i, dataset in enumerate(ergebnisse['results'][:3], 1):
  titel = dataset.get('title', {}).get('de', 'Kein Titel')
  print(f"{i}. {titel}")
```

### Schritt 2: Datensatz auswählen und untersuchen

```python
# Erstes Ergebnis auswählen
dataset_id = ergebnisse['results'][0]['id']

# Vollständige Metadaten abrufen
dataset = get_dataset(dataset_id=dataset_id)

# Wichtige Informationen überprüfen
print(f"Titel: {dataset['title']['de']}")
print(f"Beschreibung: {dataset['description']['de'][:200]}...")
print(f"Geändert: {dataset['modified']}")
print(f"Herausgeber: {dataset['publisher']['name']}")
print(f"Lizenz: {dataset.get('license', 'Nicht angegeben')}")
```

### Schritt 3: Download-URLs abrufen

```python
# Alle Distributionen abrufen
distributionen = get_dataset_distributions(dataset_id=dataset_id)

# CSV-Distribution finden
csv_dist = next(
  (d for d in distributionen
   if d.get('format', {}).get('id') == 'CSV'),
  None
)

if csv_dist:
  download_url = csv_dist['access_url']
  datei_groesse = csv_dist.get('byte_size', 'Unbekannt')
  print(f"Download-URL: {download_url}")
  print(f"Dateigröße: {datei_groesse} Bytes")
```

### Schritt 4: Vorschau vor Download

```python
# Schema zuerst prüfen
schema = preview_schema(url=download_url, format="CSV")

print("Spalten:")
for col in schema['columns']:
  print(f"  - {col['name']}: {col['type']}")

# Beispieldaten in Vorschau anzeigen
vorschau = preview_data(url=download_url, max_rows=20, format="CSV")

print(f"\nZeige {vorschau['row_count']} Beispielzeilen:")
for row in vorschau['data'][:5]:
  print(row)
```

### Schritt 5: Verifizieren und Herunterladen

```python
# Struktur auf Erwartungen überprüfen
erwartete_spalten = ["jahr", "region", "faelle"]
tatsaechliche_spalten = [c['name'] for c in schema['columns']]

if all(col in tatsaechliche_spalten for col in erwartete_spalten):
  print("✓ Schema validiert")
  # Mit vollständigem Download über download_url fortfahren
  # (Download-Implementierung abhängig von Ihrer Umgebung)
else:
  print("✗ Schema-Diskrepanz, Datensatz prüfen")
```

## Workflow 2: Semantische Suche → Verwandte Datensätze

Verwandte Daten durch intelligente Suche und Empfehlungen entdecken.

### Schritt 1: Natürlichsprachige Abfrage

```python
# Semantische Suche für natürliche Sprache verwenden
ergebnisse = semantic_search_datasets(
  natural_query="Klimawandeldaten für österreichische Städte",
  limit=5
)

print(f"{len(ergebnisse['results'])} hochrelevante Datensätze gefunden")

# Ergebnisse der semantischen Suche anzeigen
for i, dataset in enumerate(ergebnisse['results'], 1):
  titel = dataset.get('title', {}).get('de', 'Kein Titel')
  kategorien = [c.get('id') for c in dataset.get('categories', [])]
  print(f"{i}. {titel}")
  print(f"   Kategorien: {', '.join(kategorien)}")
```

### Schritt 2: Qualität analysieren

```python
# Relevantesten Datensatz auswählen
dataset_id = ergebnisse['results'][0]['id']

# Umfassende Qualitätsanalyse abrufen
qualitaet = analyze_dataset_quality(dataset_id=dataset_id)

print(f"Qualitätsbewertung: {qualitaet['score']}/100")
print(f"Vollständigkeit: {qualitaet['completeness']}")
print("\nQualitätskomponenten:")
for komponente, wert in qualitaet['components'].items():
  status = "✓" if wert else "✗"
  print(f"  {status} {komponente}")
```

### Schritt 3: Verwandte Datensätze finden

```python
# Ähnliche Datensätze entdecken
verwandt = find_related_datasets(
  dataset_id=dataset_id,
  limit=5,
  min_score=30
)

print(f"\n{len(verwandt['results'])} verwandte Datensätze gefunden:")

for dataset in verwandt['results']:
  titel = dataset.get('title', {}).get('de', 'Kein Titel')
  bewertung = dataset.get('match_score', 0)
  themen = [t.get('id') for t in dataset.get('themes', [])]

  print(f"Bewertung {bewertung}: {titel}")
  print(f"  Gemeinsame Themen: {', '.join(themen)}")
```

### Schritt 4: Datensätze vergleichen

```python
# Original- und verwandten Datensatz vergleichen
original = get_dataset(dataset_id=dataset_id)
verwandt_id = verwandt['results'][0]['id']
verwandter_dataset = get_dataset(dataset_id=verwandt_id)

# Schlüsselattribute vergleichen
print("\nVergleich:")
print(f"Original-Herausgeber: {original['publisher']['name']}")
print(f"Verwandter Herausgeber: {verwandter_dataset['publisher']['name']}")

original_themen = {c['id'] for c in original.get('categories', [])}
verwandte_themen = {c['id'] for c in verwandter_dataset.get('categories', [])}
gemeinsame_themen = original_themen & verwandte_themen

print(f"Gemeinsame Themen: {', '.join(gemeinsame_themen)}")
```

## Workflow 3: Qualitätsgefilterte Suche

Hochwertige Datensätze für den Produktionseinsatz finden.

### Schritt 1: Qualitätsbewusste Suche

```python
# Suche mit aktivierter Qualitätsverstärkung
ergebnisse = search_datasets(
  query="",
  themes=["ECON"],
  formats=["CSV", "JSON"],
  quality_boost=True,
  sort="modified+desc",
  limit=20
)
```

### Schritt 2: Nach Qualitätsmetriken filtern

```python
# Jeden Datensatz auf Qualität analysieren
hochwertige_datensaetze = []

for dataset in ergebnisse['results'][:10]:
  dataset_id = dataset['id']

  # Qualitätsanalyse abrufen
  qualitaet = analyze_dataset_quality(dataset_id=dataset_id)

  # Filter: Bewertung > 70 und alle kritischen Komponenten vorhanden
  if qualitaet['score'] > 70:
    kritisch_ok = all([
      qualitaet['components']['has_title'],
      qualitaet['components']['has_description'],
      qualitaet['components']['has_publisher'],
      qualitaet['components']['has_license']
    ])

    if kritisch_ok:
      hochwertige_datensaetze.append({
        'id': dataset_id,
        'titel': dataset.get('title', {}).get('de', 'Kein Titel'),
        'bewertung': qualitaet['score'],
        'geaendert': dataset.get('modified')
      })

print(f"{len(hochwertige_datensaetze)} hochwertige Datensätze gefunden")
```

### Schritt 3: Datenzugänglichkeit verifizieren

```python
# Sicherstellen, dass Daten tatsächlich zugänglich sind
zugaengliche_datensaetze = []

for dataset in hochwertige_datensaetze:
  # Distributionen abrufen
  distributionen = get_dataset_distributions(dataset_id=dataset['id'])

  # Auf gültige Download-URLs prüfen
  hat_csv = any(
    d.get('format', {}).get('id') == 'CSV' and d.get('access_url')
    for d in distributionen
  )

  if hat_csv:
    # URL-Zugänglichkeit durch Schema-Prüfung verifizieren
    csv_url = next(
      d['access_url'] for d in distributionen
      if d.get('format', {}).get('id') == 'CSV'
    )

    try:
      schema = preview_schema(url=csv_url, format="CSV")

      zugaengliche_datensaetze.append({
        **dataset,
        'url': csv_url,
        'spalten': len(schema['columns'])
      })

      print(f"✓ {dataset['titel']}")
      print(f"  Bewertung: {dataset['bewertung']}, Spalten: {len(schema['columns'])}")

    except Exception as e:
      print(f"✗ {dataset['titel']} - URL nicht zugänglich")

print(f"\n{len(zugaengliche_datensaetze)} Datensätze einsatzbereit")
```

## Workflow 4: Mehrsprachige Suche

Deutsche und englische Abfragen effektiv verarbeiten.

### Schritt 1: Deutsche Abfrage

```python
# Deutsche semantische Suche
de_ergebnisse = semantic_search_datasets(
  natural_query="Gesundheitsdaten aus Wien",
  limit=10
)

print("Ergebnisse der deutschen Abfrage:")
for dataset in de_ergebnisse['results'][:3]:
  print(f"- {dataset['title']['de']}")
```

### Schritt 2: Äquivalente englische Abfrage

```python
# Englische semantische Suche (gleiche Absicht)
en_ergebnisse = semantic_search_datasets(
  natural_query="health data from Vienna",
  limit=10
)

print("\nErgebnisse der englischen Abfrage:")
for dataset in en_ergebnisse['results'][:3]:
  print(f"- {dataset['title'].get('de', 'Kein Titel')}")
```

### Schritt 3: Ergebnisse vergleichen

```python
# Datensatz-IDs extrahieren
de_ids = {d['id'] for d in de_ergebnisse['results']}
en_ids = {d['id'] for d in en_ergebnisse['results']}

# Überschneidung finden
ueberschneidung = de_ids & en_ids
print(f"\nÜberschneidende Datensätze: {len(ueberschneidung)}/{len(de_ids)}")

# Spracherkennung funktionierte korrekt bei hoher Überschneidung
ueberschneidung_prozent = (len(ueberschneidung) / len(de_ids)) * 100
if ueberschneidung_prozent > 70:
  print(f"✓ Spracherkennung effektiv ({ueberschneidung_prozent:.0f}% Überschneidung)")
```

## Workflow 5: Katalog-Exploration

Alle Daten eines bestimmten Herausgebers erkunden.

### Schritt 1: Kataloge auflisten

```python
# Alle Kataloge abrufen
kataloge = list_catalogues(limit=50)

# Wien-Katalog finden
wien = next(
  c for c in kataloge
  if 'wien' in c.get('title', {}).get('de', '').lower()
)

print(f"Gefunden: {wien['title']['de']}")
print(f"Katalog-ID: {wien['id']}")
print(f"Datensatzanzahl: {wien['count']}")
```

### Schritt 2: Katalog-Metadaten abrufen

```python
# Detaillierte Kataloginformationen abrufen
katalog = get_catalogue(catalogue_id=wien['id'])

print(f"\nKatalog: {katalog['title']['de']}")
print(f"Beschreibung: {katalog.get('description', {}).get('de', 'N/A')[:200]}")
print(f"Datensätze: {katalog['count']}")
print(f"Geändert: {katalog['modified']}")
```

### Schritt 3: Katalog-Datensätze durchsuchen

```python
# Innerhalb eines bestimmten Katalogs suchen
datensaetze = search_datasets(
  query="",
  catalogues=[wien['id']],
  sort="modified+desc",
  limit=20
)

print(f"\nAktuelle Datensätze von {wien['title']['de']}:")

for i, dataset in enumerate(datensaetze['results'][:10], 1):
  titel = dataset['title']['de']
  geaendert = dataset['modified']
  kategorien = [c['id'] for c in dataset.get('categories', [])]

  print(f"{i}. {titel}")
  print(f"   Geändert: {geaendert}, Kategorien: {', '.join(kategorien)}")
```

### Schritt 4: Kategorienverteilung

```python
# Analysieren, welche Kategorien der Katalog abdeckt
kategorien_anzahl = {}

for dataset in datensaetze['results']:
  for kategorie in dataset.get('categories', []):
    kat_id = kategorie['id']
    kategorien_anzahl[kat_id] = kategorien_anzahl.get(kat_id, 0) + 1

print("\nKategorienverteilung:")
for kat, anzahl in sorted(
  kategorien_anzahl.items(),
  key=lambda x: x[1],
  reverse=True
):
  print(f"  {kat}: {anzahl} Datensätze")
```

## Best Practices

### Fehlerbehandlung

Workflows immer in try-except-Blöcke einbetten:

```python
try:
  ergebnisse = search_datasets(query="Daten", limit=10)
  # Ergebnisse verarbeiten...
except ToolError as e:
  print(f"Fehler: {e}")
  # Elegant behandeln
```

### Paginierung

Für große Ergebnismengen durch Ergebnisse paginieren:

```python
limit = 20
offset = 0
alle_ergebnisse = []

while True:
  seite = search_datasets(
    query="Umwelt",
    limit=limit,
    offset=offset
  )

  alle_ergebnisse.extend(seite['results'])

  # Prüfen, ob weitere Seiten existieren
  if offset + limit >= seite['count']:
    break

  offset += limit

print(f"{len(alle_ergebnisse)} Datensätze insgesamt abgerufen")
```

### Ergebnisse zwischenspeichern

Teure Operationen zwischenspeichern:

```python
# Qualitätsanalyseergebnisse zwischenspeichern
qualitaets_cache = {}

def qualitaet_zwischengespeichert_abrufen(dataset_id):
  if dataset_id not in qualitaets_cache:
    qualitaets_cache[dataset_id] = analyze_dataset_quality(
      dataset_id=dataset_id
    )
  return qualitaets_cache[dataset_id]
```
