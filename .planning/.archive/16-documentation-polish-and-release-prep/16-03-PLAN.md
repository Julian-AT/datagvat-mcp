<task type="checkpoint:decision" gate="blocking">
  <decision>Select approach for creating visual resources</decision>
  <context>
Need to create diagrams and potentially screenshots. Different approaches have different tradeoffs for maintenance and quality.
  </context>
  <options>
    <option id="mermaid">
      <name>Mermaid diagrams in MDX</name>
      <pros>Code-based, version-controlled, easy to update, supports dark mode automatically, accessible</pros>
      <cons>Limited styling options, may not look as polished as custom graphics</cons>
    </option>
    <option id="svg">
      <name>Custom SVG files</name>
      <pros>Full control over appearance, professional look, optimized file size</pros>
      <cons>Harder to maintain, need graphic design skills, dark mode requires separate versions</cons>
    </option>
    <option id="hybrid">
      <name>Mermaid for architecture, screenshots for UI</name>
      <pros>Best of both - maintainable diagrams, real screenshots where needed</pros>
      <cons>Mixed approach requires two workflows</cons>
    </option>
  </options>
  <resume-signal>Select: mermaid, svg, or hybrid</resume-signal>
</task>
