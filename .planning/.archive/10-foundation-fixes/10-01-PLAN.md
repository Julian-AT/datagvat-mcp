```typescript
export const config = {
  matcher: ['/((?!api|_next/static|_next/image|favicon.ico).*)'],
};
```

Create root layout at docs/app/layout.tsx (currently missing, violates Next.js requirement):

3. Export default RootLayout function that:
   - Accepts { children } prop
   - Returns <html> and <body> tags wrapping children
   - Sets lang="en" as default (locale-specific layouts override this)
   - Includes suppressHydrationWarning on html tag for theme switching

Structure:
```typescript
import type { Metadata } from 'next';

export const metadata: Metadata = {
  title: {
    default: 'Austria MCP Documentation',
    template: '%s | Austria MCP',
  },
  description: 'Comprehensive guide to the Austria MCP server for data.gv.at',
  icons: {
    icon: '/favicon.ico',
  },
};

export default function RootLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  return (
    <html lang="en" suppressHydrationWarning>
      <body>{children}</body>
    </html>
  );
}
```

This satisfies Next.js App Router requirement for root HTML structure. The [lang] layout will provide locale-specific wrapping.
