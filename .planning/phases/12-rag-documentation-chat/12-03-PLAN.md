    **Citation extraction logic**:
    ```typescript
    function extractCitations(content: string, chunks: RetrievedChunk[]): Citation[] {
      const citations: Citation[] = [];
      const regex = /\[(\d+)\]/g;
      let match;

      while ((match = regex.exec(content)) !== null) {
        const index = parseInt(match[1]) - 1;
        if (chunks[index]) {
          citations.push({
            number: match[1],
            url: chunks[index].url,
            title: chunks[index].title,
            section: chunks[index].section,
          });
        }
      }

      return citations;
    }
    ```

    **Citation rendering** (below message):
    ```tsx

        {citations.map((citation) => (
          <Link
            key={citation.url}
            href={citation.url}
            className="text-xs rounded-lg border p-2 hover:bg-fd-accent"
          >
            <span className="font-medium">[{citation.number}] {citation.title}</span>
            <span className="text-fd-muted-foreground"> — {citation.section}</span>
          </Link>
        ))}

    )}
    ```
