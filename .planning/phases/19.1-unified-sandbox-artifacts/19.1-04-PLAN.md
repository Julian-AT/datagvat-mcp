```typescript
import { e2bClient } from './client';
import { put } from '@vercel/blob';

type ExecutionResult = {
  success: boolean;
  outputs: Array<{
    type: 'stdout' | 'stderr' | 'visualization';
    content: string;
    timestamp: string;
  }>;
  error?: string;
};

export async function executeSandboxCode({
  code,
  e2bSandboxId,
  chatId,
  onOutput,
}: {
  code: string;
  e2bSandboxId?: string;
  chatId: string;
  onOutput?: (output: { type: string; content: string }) => void;
}): Promise<ExecutionResult> {
  let sandbox;
  const outputs: ExecutionResult['outputs'] = [];

  try {
    // Reuse existing sandbox or create new one (RESEARCH.md Pitfall 3)
    if (e2bSandboxId) {
      try {
        sandbox = await e2bClient.reconnectSandbox(e2bSandboxId);
      } catch {
        // Sandbox died, create new one
        sandbox = await e2bClient.createSandbox();
      }
    } else {
      sandbox = await e2bClient.createSandbox();
    }

    const result = await sandbox.runCode(code, {
      timeoutMs: 30 * 1000, // 30 second timeout (E2B-05 requirement)
      onStdout: (stdout) => {
        const output = {
          type: 'stdout' as const,
          content: stdout,
          timestamp: new Date().toISOString(),
        };
        outputs.push(output);
        onOutput?.(output);
      },
      onStderr: (stderr) => {
        const output = {
          type: 'stderr' as const,
          content: stderr,
          timestamp: new Date().toISOString(),
        };
        outputs.push(output);
        onOutput?.(output);
      },
    });

    // Handle visualizations (upload to blob storage)
    if (result.visualizations && result.visualizations.length > 0) {
      for (const viz of result.visualizations) {
        if (viz.png) {
          const blob = await put(
            `sandbox-viz/${chatId}/${Date.now()}.png`,
            Buffer.from(viz.png, 'base64'),
            {
              access: 'public',
              contentType: 'image/png',
            }
          );

          outputs.push({
            type: 'visualization',
            content: blob.url,
            timestamp: new Date().toISOString(),
          });

          onOutput?.({
            type: 'visualization',
            content: blob.url,
          });
        }
      }
    }

    return {
      success: true,
      outputs,
      e2bSandboxId: sandbox.id,
    };
  } catch (error) {
    const errorMessage = error instanceof Error ? error.message : 'Unknown execution error';

    outputs.push({
      type: 'stderr',
      content: `Error: ${errorMessage}`,
      timestamp: new Date().toISOString(),
    });

    return {
      success: false,
      outputs,
      error: errorMessage,
    };
  } finally {
    // DON'T kill sandbox - reuse it (CONTEXT.md: persistent workspace model)
    // Sandbox cleanup happens when user closes artifact or after timeout
  }
}
```

**Key features:**
- Reuse E2B sandbox via e2bSandboxId (RESEARCH.md Pitfall 3: avoid creating new sandbox per execution)
- Streaming output via onStdout/onStderr callbacks (RESEARCH.md Open Question 4: stream for better UX)
- 30-second timeout (E2B-05 requirement from Phase 18)
- Visualization upload to Vercel Blob (not base64 in DB - VIZ-03 requirement)
- Try/catch without finally kill (persistent workspace - CONTEXT.md decision)
- onOutput callback for real-time UI updates

```typescript
import { executeSandboxCode } from '@/lib/e2b/sandbox-executor';

const handleRun = useCallback(async () => {
  if (!sandbox) return;

  // Show approval UI on first run
  if (!sandbox.hasApprovedOnce) {
    updateSandbox(sandboxId, { showApproval: true });
    return;
  }

  // Set running state and clear previous outputs (CONTEXT.md: clear on run)
  updateSandbox(sandboxId, {
    isRunning: true,
    outputs: [],
    activeTab: 'output', // Auto-switch to output tab
  });

  try {
    const result = await executeSandboxCode({
      code: sandbox.code,
      e2bSandboxId: sandbox.e2bSandboxId,
      chatId,
      onOutput: (output) => {
        // Stream outputs to UI in real-time
        updateSandbox(sandboxId, (current) => ({
          ...current,
          outputs: [
            ...(current.outputs || []),
            {
              type: output.type as 'stdout' | 'stderr' | 'visualization',
              content: output.content,
              timestamp: new Date().toISOString(),
            },
          ],
        }));
      },
    });

    // Update sandbox with final state
    updateSandbox(sandboxId, {
      isRunning: false,
      outputs: result.outputs,
      e2bSandboxId: result.e2bSandboxId,
    });

    // Save to DB
    await saveSandboxState({
      sandboxId,
      messageId,
      chatId,
      title: sandbox.title,
      code: sandbox.code,
      outputs: result.outputs,
      hasApprovedOnce: sandbox.hasApprovedOnce,
      e2bSandboxId: result.e2bSandboxId,
    });
  } catch (error) {
    console.error('Execution error:', error);
    updateSandbox(sandboxId, {
      isRunning: false,
      outputs: [
        {
          type: 'stderr',
          content: `Execution failed: ${error instanceof Error ? error.message : 'Unknown error'}`,
          timestamp: new Date().toISOString(),
        },
      ],
    });
  }
}, [sandbox, sandboxId, chatId, messageId, updateSandbox]);
```

- Auto-switch to output tab when running (better UX)
- Stream outputs via onOutput callback (real-time display)
- Save final state to DB including e2bSandboxId for reuse
- Handle errors gracefully with stderr output
