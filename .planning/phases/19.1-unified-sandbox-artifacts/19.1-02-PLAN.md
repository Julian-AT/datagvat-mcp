```typescript
const containerRef = useRef<HTMLDivElement>(null);
const editorRef = useRef<EditorView | null>(null);

useEffect(() => {
  if (containerRef.current && !editorRef.current) {
    const startState = EditorState.create({
      doc: code,
      extensions: [
        basicSetup,
        python(),
        oneDark,
        EditorView.updateListener.of((update) => {
          if (update.docChanged) {
            onCodeChange(update.state.doc.toString());
          }
        }),
      ],
    });

    editorRef.current = new EditorView({
      state: startState,
      parent: containerRef.current,
    });
  }

  return () => {
    if (editorRef.current) {
      editorRef.current.destroy();
      editorRef.current = null;
    }
  };
}, []);
```

```typescript
import Ansi from 'ansi-to-react';

{outputs.map((output, idx) => {
  if (output.type === 'stdout') {
    return (
      <div key={idx} className="font-mono text-sm">
        <Ansi>{output.content}</Ansi>
      </div>
    );
  }

  if (output.type === 'stderr') {
    return (
      <div key={idx} className="font-mono text-sm text-red-600 bg-red-50 p-2 rounded">
        <Ansi>{output.content}</Ansi>
      </div>
    );
  }

  if (output.type === 'visualization') {
    return (
      <div key={idx} className="my-4">
        <img
          src={output.content} // Blob URL
          alt="Visualization"
          className="max-w-[400px] cursor-pointer rounded border"
          onClick={() => openFullscreen(output.content)}
        />
      </div>
    );
  }
})}
```
