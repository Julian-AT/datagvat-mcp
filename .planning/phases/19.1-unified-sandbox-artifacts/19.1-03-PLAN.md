Create useSandboxes hook following use-artifact.ts pattern (RESEARCH.md Pattern 4):

```typescript
import { useCallback, useMemo } from 'react';
import useSWR from 'swr';

export type UISandbox = {
  sandboxId: string;
  title: string;
  code: string;
  outputs: Array<{
    type: 'stdout' | 'stderr' | 'visualization';
    content: string;
    timestamp: string;
  }>;
  hasApprovedOnce: boolean;
  e2bSandboxId?: string;
  isRunning: boolean;
  activeTab: 'code' | 'output';
  showApproval: boolean;
};

export function useSandboxes() {
  const { data: sandboxes, mutate: setSandboxes } = useSWR<UISandbox[]>(
    'sandboxes',
    null,
    { fallbackData: [] }
  );

  const openSandbox = useCallback(
    (sandbox: UISandbox) => {
      setSandboxes((current) => [...(current || []), sandbox]);
    },
    [setSandboxes]
  );

  const closeSandbox = useCallback(
    (sandboxId: string) => {
      setSandboxes((current) =>
        (current || []).filter((s) => s.sandboxId !== sandboxId)
      );
    },
    [setSandboxes]
  );

  const updateSandbox = useCallback(
    (sandboxId: string, updates: Partial<UISandbox>) => {
      setSandboxes((current) =>
        (current || []).map((s) =>
          s.sandboxId === sandboxId ? { ...s, ...updates } : s
        )
      );
    },
    [setSandboxes]
  );

  const activeSandbox = useMemo(
    () => sandboxes?.[sandboxes.length - 1],
    [sandboxes]
  );

  return {
    sandboxes: sandboxes || [],
    openSandbox,
    closeSandbox,
    updateSandbox,
    activeSandbox,
  };
}
```

**Key features:**
- Array state for multiple sandboxes (CONTEXT.md decision)
- openSandbox/closeSandbox/updateSandbox methods
- activeSandbox = most recent (last in array)
- SWR with fallbackData: [] for hydration safety

```typescript
import * as Tabs from '@radix-ui/react-tabs';
import { useCallback, useEffect, useState } from 'react';
import { SandboxCodeTab } from './sandbox-code-tab';
import { SandboxOutputTab } from './sandbox-output-tab';
import { useSandboxes } from '@/hooks/use-sandbox';
import { saveSandboxState, getSandboxState } from '@/lib/db/queries';

type SandboxArtifactProps = {
  sandboxId: string;
  chatId: string;
  messageId: string;
};

export function SandboxArtifact({
  sandboxId,
  chatId,
  messageId,
}: SandboxArtifactProps) {
  const { sandboxes, updateSandbox } = useSandboxes();
  const sandbox = sandboxes.find((s) => s.sandboxId === sandboxId);

  const [activeTab, setActiveTab] = useState<'code' | 'output'>('code');

  // Load sandbox state on mount (page refresh restoration)
  useEffect(() => {
    if (!sandbox) {
      getSandboxState(sandboxId).then((state) => {
        if (state) {
          updateSandbox(sandboxId, {
            code: state.code,
            outputs: state.outputs || [],
            hasApprovedOnce: state.hasApprovedOnce || false,
            e2bSandboxId: state.e2bSandboxId || undefined,
          });
        }
      });
    }
  }, [sandboxId]);

  const handleCodeChange = useCallback(
    (code: string) => {
      updateSandbox(sandboxId, { code });
      // Auto-save to DB
      saveSandboxState({
        sandboxId,
        messageId,
        chatId,
        title: sandbox?.title || 'Sandbox',
        code,
        outputs: sandbox?.outputs,
        hasApprovedOnce: sandbox?.hasApprovedOnce,
        e2bSandboxId: sandbox?.e2bSandboxId,
      });
    },
    [sandboxId, chatId, messageId, sandbox, updateSandbox]
  );

  const handleRun = useCallback(() => {
    if (!sandbox?.hasApprovedOnce) {
      updateSandbox(sandboxId, { showApproval: true });
      return;
    }
    // Execution will be wired in Plan 04
    console.log('Run code:', sandbox.code);
  }, [sandbox, sandboxId, updateSandbox]);

  const handleApprove = useCallback(() => {
    updateSandbox(sandboxId, {
      hasApprovedOnce: true,
      showApproval: false,
    });
    // Then execute
    handleRun();
  }, [sandboxId, updateSandbox, handleRun]);

  const handleDeny = useCallback(() => {
    // CONTEXT.md decision: denial closes sandbox
    // Will be wired to closeSandbox in Plan 05
    console.log('Execution denied - sandbox will close');
  }, []);

  if (!sandbox) {
    return <div>Loading sandbox...</div>;
  }

  return (
    <div className="border rounded-lg overflow-hidden">
      <div className="bg-muted px-4 py-2 border-b">
        <h3 className="font-medium">{sandbox.title}</h3>
      </div>

      <Tabs.Root value={activeTab} onValueChange={(v) => setActiveTab(v as 'code' | 'output')}>
        <Tabs.List className="flex border-b">
          <Tabs.Trigger
            value="code"
            className="px-4 py-2 data-[state=active]:border-b-2 data-[state=active]:border-primary"
          >
            Code
          </Tabs.Trigger>
          <Tabs.Trigger
            value="output"
            className="px-4 py-2 data-[state=active]:border-b-2 data-[state=active]:border-primary"
          >
            Output
          </Tabs.Trigger>
        </Tabs.List>

        <Tabs.Content value="code" className="p-4">
          <SandboxCodeTab
            code={sandbox.code}
            onCodeChange={handleCodeChange}
            onRun={handleRun}
            showApproval={sandbox.showApproval}
            onApprove={handleApprove}
            onDeny={handleDeny}
            hasApprovedOnce={sandbox.hasApprovedOnce}
            isRunning={sandbox.isRunning}
          />
        </Tabs.Content>

        <Tabs.Content value="output" className="p-4">
          <SandboxOutputTab outputs={sandbox.outputs} />
        </Tabs.Content>
      </Tabs.Root>
    </div>
  );
}
```

- Page refresh restoration via getSandboxState on mount
- Auto-save code changes to DB via saveSandboxState
- Approval flow: first run shows approval UI, deny closes sandbox
- Placeholder execution logic (wired in Plan 04)
