```typescript
import { db } from '@/db';
import { messages } from '@/db/schema';
import { eq } from 'drizzle-orm';
import type { Sandbox } from '@e2b/code-interpreter';
import { createE2BClient } from '@/lib/mcp/e2b-client';

export interface TrackedSandbox {
  sandbox: Sandbox;
  messageId: number;
}

export async function createTrackedSandbox(
  messageId: number
): Promise<TrackedSandbox> {
  const e2bClient = createE2BClient({
    apiKey: process.env.E2B_API_KEY || '',
    timeoutMs: 60 * 60 * 1000, // 1 hour (EXEC-06)
  });

  const sandbox = await e2bClient.createSandbox();

  // Track in messages table where sandbox_id column exists
  await db
    .update(messages)
    .set({
      sandboxId: sandbox.sandboxId,
    })
    .where(eq(messages.id, messageId));

  return { sandbox, messageId };
}

export async function cleanupSandbox(messageId: number): Promise<void> {
  const message = await db.query.messages.findFirst({
    where: eq(messages.id, messageId),
  });

  if (!message?.sandboxId) {
    return; // No active sandbox
  }

  // E2B sandboxes auto-terminate on timeout, just clear DB
  await db
    .update(messages)
    .set({
      sandboxId: null,
    })
    .where(eq(messages.id, messageId));
}
```

Satisfies requirements:
- SEC-04: Each conversation uses isolated sandbox (tracked per message for code execution)
- EXEC-06: 1-hour timeout enforced via timeoutMs
- Database tracking: sandbox_id column in messages table (line 79 of schema.ts)

IMPORTANT: sandbox_id is stored in messages table, NOT conversations table. This correctly tracks which message triggered sandbox creation.

WHY message-level tracking: Code execution is message-specific. Multiple messages in same conversation may use different sandboxes.

**docs/app/api/cron/cleanup-sandboxes/route.ts:**
```typescript
import { NextResponse } from 'next/server';
import { db } from '@/db';
import { messages } from '@/db/schema';
import { sql } from 'drizzle-orm';

export const runtime = 'nodejs';
export const dynamic = 'force-dynamic';
export const maxDuration = 60; // 60 seconds max execution

export async function GET(request: Request) {
  // Verify cron secret (security)
  const authHeader = request.headers.get('authorization');
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    return new Response('Unauthorized', { status: 401 });
  }

  try {
    // Find sandboxes older than 1 hour (track via message created_at timestamp)
    const orphanedSandboxes = await db
      .select()
      .from(messages)
      .where(
        sql`sandbox_id IS NOT NULL
            AND created_at < NOW() - INTERVAL '1 hour'`
      );

    // Cleanup (E2B sandboxes auto-terminate on timeout, just clear DB)
    const cleaned = await db
      .update(messages)
      .set({
        sandboxId: null,
      })
      .where(
        sql`sandbox_id IS NOT NULL
            AND created_at < NOW() - INTERVAL '1 hour'`
      )
      .returning();

    console.log(`Cleaned ${cleaned.length} orphaned sandboxes`);

    return NextResponse.json({
      cleaned: cleaned.length,
      timestamp: new Date().toISOString(),
    });
  } catch (error) {
    console.error('Sandbox cleanup failed:', error);
    return NextResponse.json(
      { error: 'Cleanup failed' },
      { status: 500 }
    );
  }
}
```

Reference pattern: 15-RESEARCH.md "Vercel Cron for Sandbox Cleanup"

Satisfies requirement EXEC-06: Automatic cleanup after 1 hour.

IMPORTANT: Uses messages.created_at as the age reference since messages table doesn't have a separate sandbox_created_at column. This is sufficient because sandbox_id is only set when sandbox is created.

WHY 1-hour timeout: Balances user workflow continuity (long data analysis sessions) with resource conservation (E2B charges per second).
