Create `docs/db/schema.ts` extending Vercel's AI SDK chat schema:

```typescript
import { pgTable, serial, text, timestamp, jsonb, varchar, index } from 'drizzle-orm/pg-core';

// AI SDK MessagePart types (matches UIMessage.parts)
export type MessagePart =
  | { type: 'text'; text: string }
  | { type: 'tool-call'; toolCallId: string; toolName: string; args: Record<string, unknown> }
  | { type: 'tool-result'; toolCallId: string; toolName: string; result: unknown }
  | { type: 'file'; filename: string; mediaType: string; url: string }; // Blob URL only

export const conversations = pgTable('conversations', {
  id: serial('id').primaryKey(),
  userId: text('user_id').notNull(), // From better-auth session (Phase 14-02)
  createdAt: timestamp('created_at', { mode: 'date' }).defaultNow().notNull(),
  updatedAt: timestamp('updated_at', { mode: 'date' }).$onUpdate(() => new Date()),
  title: text('title'),
}, (table) => [
  index('user_conversations_idx').on(table.userId),
  index('conversation_created_idx').on(table.createdAt),
]);

export const messages = pgTable('messages', {
  id: serial('id').primaryKey(),
  conversationId: serial('conversation_id').references(() => conversations.id, { onDelete: 'cascade' }).notNull(),
  role: varchar('role', { length: 20 }).notNull(), // 'user' | 'assistant' | 'system'
  parts: jsonb('parts').$type<MessagePart[]>().notNull(),
  createdAt: timestamp('created_at', { mode: 'date' }).defaultNow().notNull(),

  // Security and MCP metadata (custom columns beyond AI SDK base)
  executionStatus: varchar('execution_status', { length: 20 }).default('pending'), // 'pending' | 'approved' | 'executed' | 'rejected'
  sandboxId: text('sandbox_id'), // Daytona workspace reference (Phase 15)
  mcpSource: text('mcp_source'), // Which MCP server tool came from
  metadata: jsonb('metadata'), // Flexible additional data
}, (table) => [
  index('conversation_messages_idx').on(table.conversationId),
  index('message_created_idx').on(table.createdAt),
  // GIN index with jsonb_path_ops for fast JSONB containment queries
  index('message_parts_gin_idx').on(table.parts).using('gin').op('jsonb_path_ops'),
]);
```

Create `docs/db/index.ts` for database client:

```typescript
import { neon } from '@neondatabase/serverless';
import { drizzle } from 'drizzle-orm/neon-http';
import * as schema from './schema';

// Enable HTTP-only mode for edge compatibility
import { neonConfig } from '@neondatabase/serverless';
neonConfig.poolQueryViaFetch = true;

const sql = neon(process.env.DATABASE_URL!);
export const db = drizzle(sql, { schema });
```

```typescript
  },
});
```
