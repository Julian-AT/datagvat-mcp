```typescript
if (type === 'dynamic-tool') {
  const toolPart = part as ToolUIPart;

  // Handle execute-python approval flow
  if (toolPart.type === 'execute-python' && toolPart.state === 'approval-requested') {
    const input = toolPart.input as { code: string; files?: Array<{ path: string; content: string }> };

    return (
      <ToolApproval
        key={toolPart.toolCallId}
        toolCallId={toolPart.toolCallId}
        toolName="execute-python"
        code={input.code}
        files={input.files}
        onApprove={async (toolCallId, approved, reason) => {
          // Persist approval to database
          await saveToolApproval({
            toolCallId,
            chatId,
            userId: /* extract from session or pass as prop */,
            toolName: 'execute-python',
            approved,
            deniedReason: reason,
            code: input.code,
          });

          // Send approval response to AI SDK
          addToolApprovalResponse({
            id: toolCallId,
            approved,
            reason,
          });
        }}
      />
    );
  }

  // Existing dynamic-tool rendering for other states...
}
```
