**Pattern to adopt:**
```typescript
export function Message({ message }: { message: ChatMessage }) {
  return (
    <div className="message" data-role={message.role}>
      {message.parts.map((part, index) => {
        const key = `${message.id}-part-${index}`;

        if (part.type === "text") {
          return (
            <MessageContent key={key}>
              {part.text}
            </MessageContent>
          );
        }

        if (part.type === "tool-call") {
          return <ToolCallDisplay key={key} toolCall={part} />;
        }

        if (part.type === "tool-result") {
          return <ToolResultDisplay key={key} result={part} />;
        }

        if (part.type === "file") {
          return (
            <PreviewAttachment
              key={key}
              attachment={{
                url: part.url,
                name: part.filename,
                contentType: part.mediaType,
              }}
            />
          );
        }

        if (part.type === "visualization") {
          return (
            <Visualization
              key={key}
              format={part.format}
              url={part.url}
              metadata={part.metadata}
            />
          );
        }

        return null;
      })}
    </div>
  );
}
```

- Map through `message.parts` array (not separate fields)

- Use stable keys (`${message.id}-part-${index}`)
