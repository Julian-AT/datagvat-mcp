Create a new component for grid layout:

```typescript
'use client';

import { useState } from 'react';
import { Download, Maximize2 } from 'lucide-react';
import { Button } from '@/components/ui/button';
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
} from '@/components/ui/dialog';
import { cn } from '@/lib/utils';

export interface VisualizationItem {
  url: string;
  format: 'png' | 'svg' | 'html';
  title?: string;
}

interface VisualizationGridProps {
  visualizations: VisualizationItem[];
  className?: string;
}

export function VisualizationGrid({ visualizations, className }: VisualizationGridProps) {
  const [fullscreenViz, setFullscreenViz] = useState<VisualizationItem | null>(null);
  const [loadedVizs, setLoadedVizs] = useState<Set<number>>(new Set());

  const count = visualizations.length;
  
  // Determine grid layout based on count
  const getGridClasses = () => {
    switch (count) {
      case 1:
        return 'grid-cols-1';
      case 2:
        return 'grid-cols-1 md:grid-cols-2';
      case 3:
        return 'grid-cols-1 md:grid-cols-2 lg:grid-cols-3';
      case 4:
        return 'grid-cols-1 md:grid-cols-2';
      default:
        return 'grid-cols-1 md:grid-cols-2 lg:grid-cols-3';
    }
  };

  const handleLoad = (index: number) => {
    setLoadedVizs(prev => new Set(prev).add(index));
  };

  const handleDownload = async (viz: VisualizationItem) => {
    try {
      const response = await fetch(viz.url);
      const blob = await response.blob();
      const url = window.URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `visualization-${Date.now()}.${viz.format}`;
      document.body.appendChild(a);
      a.click();
      window.URL.revokeObjectURL(url);
      document.body.removeChild(a);
    } catch (err) {
      console.error('Failed to download:', err);
    }
  };

  const renderVisualization = (viz: VisualizationItem, index: number, fullscreen = false) => {
    const isLoaded = loadedVizs.has(index);
    
    if (viz.format === 'html') {
      return (
        <iframe
          src={viz.url}
          className={cn(
            'w-full bg-white',
            fullscreen ? 'h-[80vh]' : 'h-[300px]',
            !isLoaded && 'hidden'
          )}
          onLoad={() => handleLoad(index)}
          title={viz.title || `Visualization ${index + 1}`}
          sandbox="allow-scripts allow-same-origin"
        />
      );
    }

    return (
      <img
        src={viz.url}
        alt={viz.title || `Visualization ${index + 1}`}
        className={cn(
          'w-full h-auto rounded-md bg-white',
          fullscreen ? 'max-h-[80vh] object-contain' : '',
          !isLoaded && 'hidden'
        )}
        onLoad={() => handleLoad(index)}
        loading="lazy"
      />
    );
  };

  if (count === 0) return null;

  return (
    <>
      <div className={cn('grid gap-4', getGridClasses(), className)}>
        {visualizations.map((viz, index) => (
          <div
            key={`${viz.url}-${index}`}
            className="group relative overflow-hidden rounded-lg border border-border bg-white"
          >
            {/* Loading placeholder */}
            {!loadedVizs.has(index) && (
              <div className="flex items-center justify-center h-[300px] bg-muted">
                <div className="animate-pulse text-muted-foreground">
                  Loading visualization...
                </div>
              </div>
            )}

            {/* Visualization */}
            {renderVisualization(viz, index)}

            {/* Actions overlay */}
            <div className="absolute top-2 right-2 flex gap-1 opacity-0 group-hover:opacity-100 transition-opacity">
              <Button
                size="sm"
                variant="secondary"
                className="h-7 w-7 p-0 bg-white/90"
                onClick={() => setFullscreenViz(viz)}
              >
                <Maximize2 className="h-3.5 w-3.5" />
              </Button>
              <Button
                size="sm"
                variant="secondary"
                className="h-7 w-7 p-0 bg-white/90"
                onClick={() => handleDownload(viz)}
              >
                <Download className="h-3.5 w-3.5" />
              </Button>
            </div>

            {/* Title if provided */}
            {viz.title && (
              <div className="absolute bottom-0 left-0 right-0 bg-gradient-to-t from-black/60 to-transparent p-2">
                <span className="text-xs text-white font-medium">{viz.title}</span>
              </div>
            )}
          </div>
        ))}
      </div>

      {/* Fullscreen dialog */}
      <Dialog open={!!fullscreenViz} onOpenChange={() => setFullscreenViz(null)}>
        <DialogContent className="max-w-[95vw] max-h-[95vh] w-full h-full p-4">
          <DialogHeader>
            <DialogTitle>Visualization</DialogTitle>
          </DialogHeader>
          {fullscreenViz && (
            <div className="flex items-center justify-center h-full">
              {renderVisualization(
                fullscreenViz,
                visualizations.findIndex(v => v.url === fullscreenViz.url),
                true
              )}
            </div>
          )}
        </DialogContent>
      </Dialog>
    </>
  );
}
```

- Lazy loading via `loading="lazy"` on images
- Loading state with pulse animation

- Supports PNG, SVG, HTML formats
- Sandboxed iframe for HTML (security)

Add an IntersectionObserver-based lazy loading hook:

```typescript
import { useEffect, useRef, useState } from 'react';

// Hook for intersection observer lazy loading
function useLazyLoad(index: number) {
  const [isVisible, setIsVisible] = useState(false);
  const [hasLoaded, setHasLoaded] = useState(false);
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsVisible(true);
          observer.disconnect();
        }
      },
      { rootMargin: '100px' } // Start loading 100px before visible
    );

    if (ref.current) {
      observer.observe(ref.current);
    }

    return () => observer.disconnect();
  }, []);

  const markLoaded = () => setHasLoaded(true);

  return { ref, isVisible, hasLoaded, markLoaded };
}
```
