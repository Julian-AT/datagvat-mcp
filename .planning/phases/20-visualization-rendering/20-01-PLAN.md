```typescript
export async function uploadHtmlFromBase64(
  base64Data: string,
  filename: string,
  chatId: string
): Promise<string> {
  // Handle both data URI format and raw base64
  const isDataUri = base64Data.startsWith('data:');
  const byteString = isDataUri ? atob(base64Data.split(',')[1]) : atob(base64Data);
  
  const ab = new ArrayBuffer(byteString.length);
  const ia = new Uint8Array(ab);

  for (let i = 0; i < byteString.length; i++) {
    ia[i] = byteString.charCodeAt(i);
  }

  const blob = new Blob([ab], { type: 'text/html' });
  
  return uploadVisualization(blob, filename, chatId, 'text/html');
}
```
