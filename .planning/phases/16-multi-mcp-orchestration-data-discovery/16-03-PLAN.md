**Expected schema from Phase 14-01:**
- Table: messages
- Columns: id (uuid), conversationId (uuid FK), role (text), parts (jsonb), createdAt (timestamp)
- Parts format: AI SDK 6 message parts (text, tool_call, tool_result)

**Implementation reference:**
```typescript
import { db } from './db'; // Drizzle instance
import { messages } from './schema';
import { eq, desc } from 'drizzle-orm';

export async function getMessages(conversationId: string, limit = 50) {
  const results = await db
    .select()
    .from(messages)
    .where(eq(messages.conversationId, conversationId))
    .orderBy(desc(messages.createdAt))
    .limit(limit);

  // Return in chronological order (reverse DESC query)
  return results.reverse();
}

export async function saveMessages(conversationId: string, msgs: Message[]) {
  const records = msgs.map(msg => ({
    id: crypto.randomUUID(),
    conversationId,
    role: msg.role,
    parts: msg.parts, // JSONB
    createdAt: new Date(),
  }));

  await db.insert(messages).values(records);
}
```
