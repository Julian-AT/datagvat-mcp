import pandas as pd
from typing import Optional

def analyze_air_quality(csv_url: str, station_filter: Optional[str] = None) -> pd.DataFrame:
    """
    Analyze PM2.5 air quality measurements from data.gv.at.

    Args:
        csv_url: Distribution download URL
        station_filter: Optional station ID to filter results

    Returns:
        DataFrame with daily mean PM2.5 by station

    Raises:
        ValueError: If CSV missing required columns or cannot be parsed
    """
    try:
        df = pd.read_csv(csv_url)

        # Verify required columns exist (from schema)
        required_cols = ['Date', 'PM2.5', 'Station_ID']
        if not all(col in df.columns for col in required_cols):
            raise ValueError(f"Missing required columns. Found: {df.columns.tolist()}")

        # Convert date column using exact name from schema
        df['Date'] = pd.to_datetime(df['Date'])

        # Filter if requested
        if station_filter:
            df = df[df['Station_ID'] == station_filter]

        # Handle missing values (completeness: 87%)
        df_clean = df.dropna(subset=['PM2.5'])

        # Calculate daily means
        return df_clean.groupby(['Date', 'Station_ID'])['PM2.5'].mean().reset_index()

    except Exception as e:
        raise ValueError(f"Analysis failed: {e}")
